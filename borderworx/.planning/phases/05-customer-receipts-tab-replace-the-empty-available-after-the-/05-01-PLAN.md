    ```typescript
    import type { PrismaClient } from '@prisma/client';
    import type { ReceiptConfig, UpsertReceiptConfigData } from '../../domain/entities/receipt-config.entity.js';
    import type { ReceiptConfigRepository } from '../../domain/repositories/receipt-config.repository.js';
    import { ReceiptConfigMapper } from './receipt-config.mapper.js';

    export class PrismaReceiptConfigRepository implements ReceiptConfigRepository {
      constructor(private readonly prisma: PrismaClient) {}

      async findByCustomerId(customerId: string): Promise<ReceiptConfig | null> {
        const record = await this.prisma.customerReceiptConfig.findUnique({ where: { customerId } });
        return record ? ReceiptConfigMapper.toDomain(record) : null;
      }

      async upsert(customerId: string, data: UpsertReceiptConfigData): Promise<ReceiptConfig> {
        const record = await this.prisma.customerReceiptConfig.upsert({
          where: { customerId },
          create: {
            customerId,
            locationsEnabled: data.locationsEnabled ?? false,
            suppliersEnabled: data.suppliersEnabled ?? false,
            fuelSurchargePercentage: data.fuelSurchargePercentage ?? 0,
            enableSuggestedPutaway: data.enableSuggestedPutaway ?? false,
            receiveAgainstAsn: data.receiveAgainstAsn ?? false,
            defaultQuantityOne: data.defaultQuantityOne ?? false,
            allowMoveUnconfirmedInventory: data.allowMoveUnconfirmedInventory ?? false,
            useBlindReceipts: data.useBlindReceipts ?? false,
            displayWeightField: data.displayWeightField ?? false,
          },
          update: {
            ...(data.locationsEnabled !== undefined && { locationsEnabled: data.locationsEnabled }),
            ...(data.suppliersEnabled !== undefined && { suppliersEnabled: data.suppliersEnabled }),
            ...(data.fuelSurchargePercentage !== undefined && { fuelSurchargePercentage: data.fuelSurchargePercentage }),
            ...(data.enableSuggestedPutaway !== undefined && { enableSuggestedPutaway: data.enableSuggestedPutaway }),
            ...(data.receiveAgainstAsn !== undefined && { receiveAgainstAsn: data.receiveAgainstAsn }),
            ...(data.defaultQuantityOne !== undefined && { defaultQuantityOne: data.defaultQuantityOne }),
            ...(data.allowMoveUnconfirmedInventory !== undefined && { allowMoveUnconfirmedInventory: data.allowMoveUnconfirmedInventory }),
            ...(data.useBlindReceipts !== undefined && { useBlindReceipts: data.useBlindReceipts }),
            ...(data.displayWeightField !== undefined && { displayWeightField: data.displayWeightField }),
          },
        });
        return ReceiptConfigMapper.toDomain(record);
      }
    }
    ```
