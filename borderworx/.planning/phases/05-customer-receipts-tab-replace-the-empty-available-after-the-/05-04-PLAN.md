      fuelSurchargePercentage: '0.00',

    export function receiptFormValuesToRequest(v: ReceiptConfigFormValues): ReceiptConfigRequest {
      const parsed = Number(v.fuelSurchargePercentage);
      return {
        locationsEnabled: v.locationsEnabled,
        suppliersEnabled: v.suppliersEnabled,
        fuelSurchargePercentage: Number.isFinite(parsed) ? parsed : 0,
        enableSuggestedPutaway: v.enableSuggestedPutaway,
        receiveAgainstAsn: v.receiveAgainstAsn,
        defaultQuantityOne: v.defaultQuantityOne,
        allowMoveUnconfirmedInventory: v.allowMoveUnconfirmedInventory,
        useBlindReceipts: v.useBlindReceipts,
        displayWeightField: v.displayWeightField,
      };
    }
